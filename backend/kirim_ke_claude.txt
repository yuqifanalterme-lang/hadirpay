<?php

namespace App\Controllers\Api\V1;

use App\Controllers\BaseController;
use App\Libraries\ApiResponse;
use App\Libraries\JwtService;

class Auth extends BaseController
{
    private const ACCESS_TTL  = 3600;           // 60 menit
    private const REFRESH_TTL = 7 * 24 * 3600;  // 7 hari

    private const MENU = [
        'admin_hrd' => ['dashboard', 'absensi', 'pengajuan', 'karyawan', 'penggajian', 'akuntansi', 'laporan', 'pengaturan'],
        'karyawan'  => ['dashboard', 'absensi', 'pengajuan', 'slip_gaji'],
        'keuangan'  => ['dashboard', 'absensi', 'pengajuan', 'penggajian', 'akuntansi', 'laporan'],
        'pemilik'   => ['dashboard', 'karyawan', 'penggajian', 'akuntansi', 'laporan'],
    ];

    public function login()
    {
        $in = $this->request->getJSON(true) ?? [];

        $errors = [];
        if (empty($in['email']))    $errors['email']    = ['Email wajib diisi'];
        if (empty($in['password'])) $errors['password'] = ['Password wajib diisi'];
        if ($errors) {
            return ApiResponse::error('VALIDATION_ERROR', 'Data tidak valid', $errors, 422);
        }

        $db   = db_connect();
        $user = $db->table('users')->where('email', trim($in['email']))->get()->getRowArray();

        if (! $user || ! $user['aktif'] || ! password_verify($in['password'], $user['password_hash'])) {
            return ApiResponse::error('INVALID_CREDENTIALS', 'Email atau password salah', null, 401);
        }

        $db->table('users')->where('id', $user['id'])->update(['last_login_at' => date('Y-m-d H:i:s')]);

        return ApiResponse::success([
            'access_token'  => JwtService::buat($user, self::ACCESS_TTL),
            'refresh_token' => $this->buatRefreshToken((int) $user['id']),
            'expires_in'    => self::ACCESS_TTL,
            'user'          => $this->profil((int) $user['id']),
        ], 'Login berhasil');
    }

    public function refresh()
    {
        $in = $this->request->getJSON(true) ?? [];
        if (empty($in['refresh_token'])) {
            return ApiResponse::error('VALIDATION_ERROR', 'Data tidak valid', ['refresh_token' => ['Refresh token wajib diisi']], 422);
        }

        $db  = db_connect();
        $row = $db->table('refresh_tokens')
            ->where('token_hash', hash('sha256', $in['refresh_token']))
            ->where('revoked_at', null)
            ->where('expires_at >', date('Y-m-d H:i:s'))
            ->get()->getRowArray();

        $user = $row ? $db->table('users')->where('id', $row['user_id'])->where('aktif', 1)->get()->getRowArray() : null;
        if (! $user) {
            return ApiResponse::error('UNAUTHENTICATED', 'Refresh token tidak valid atau kedaluwarsa', null, 401);
        }

        return ApiResponse::success([
            'access_token' => JwtService::buat($user, self::ACCESS_TTL),
            'expires_in'   => self::ACCESS_TTL,
        ], 'Token diperbarui');
    }

    public function logout()
    {
        db_connect()->table('refresh_tokens')
            ->where('user_id', JwtService::$user['id'])
            ->where('revoked_at', null)
            ->update(['revoked_at' => date('Y-m-d H:i:s')]);

        return ApiResponse::success(null, 'Logout berhasil');
    }

    public function me()
    {
        $profil = $this->profil((int) JwtService::$user['id']);
        $profil['menu_access'] = self::MENU[$profil['role']] ?? ['dashboard'];

        return ApiResponse::success($profil);
    }

    public function password()
    {
        $in   = $this->request->getJSON(true) ?? [];
        $user = JwtService::$user;

        $errors = [];
        if (empty($in['password_lama'])) {
            $errors['password_lama'] = ['Password lama wajib diisi'];
        } elseif (! password_verify($in['password_lama'], $user['password_hash'])) {
            $errors['password_lama'] = ['Password lama salah'];
        }
        if (empty($in['password_baru']) || strlen($in['password_baru']) < 8) {
            $errors['password_baru'] = ['Password baru minimal 8 karakter'];
        }
        if (($in['konfirmasi_password'] ?? null) !== ($in['password_baru'] ?? '')) {
            $errors['konfirmasi_password'] = ['Konfirmasi password tidak sama'];
        }
        if ($errors) {
            return ApiResponse::error('VALIDATION_ERROR', 'Data tidak valid', $errors, 422);
        }

        $db = db_connect();
        $db->table('users')->where('id', $user['id'])->update([
            'password_hash' => password_hash($in['password_baru'], PASSWORD_DEFAULT),
            'updated_at'    => date('Y-m-d H:i:s'),
        ]);
        $db->table('refresh_tokens')->where('user_id', $user['id'])->where('revoked_at', null)
            ->update(['revoked_at' => date('Y-m-d H:i:s')]);

        return ApiResponse::success(null, 'Password berhasil diubah');
    }

    // ---------- helper ----------

    private function buatRefreshToken(int $userId): string
    {
        $token = bin2hex(random_bytes(32));
        db_connect()->table('refresh_tokens')->insert([
            'user_id'    => $userId,
            'token_hash' => hash('sha256', $token),
            'expires_at' => date('Y-m-d H:i:s', time() + self::REFRESH_TTL),
            'created_at' => date('Y-m-d H:i:s'),
        ]);
        return $token;
    }

    private function profil(int $userId): array
    {
        $row = db_connect()->table('users u')
            ->select('u.id, u.karyawan_id, u.nama, u.email, u.role, j.nama AS jabatan')
            ->join('karyawan k', 'k.id = u.karyawan_id', 'left')
            ->join('jabatan j', 'j.id = k.jabatan_id', 'left')
            ->where('u.id', $userId)
            ->get()->getRowArray();

        $kata    = preg_split('/\s+/', trim($row['nama']));
        $inisial = strtoupper(mb_substr($kata[0], 0, 1) . (isset($kata[1]) ? mb_substr($kata[1], 0, 1) : ''));

        return [
            'id'          => (int) $row['id'],
            'karyawan_id' => $row['karyawan_id'] !== null ? (int) $row['karyawan_id'] : null,
            'nama'        => $row['nama'],
            'email'       => $row['email'],
            'role'        => $row['role'],
            'jabatan'     => $row['jabatan'] ?? ($row['role'] === 'pemilik' ? 'Pemilik' : null),
            'inisial'     => $inisial,
            'foto_url'    => null,
        ];
    }
}

<?php

namespace App\Controllers\Api\V1;

use App\Controllers\BaseController;
use App\Libraries\ApiResponse;
use App\Models\JabatanModel;
use CodeIgniter\Database\Exceptions\DatabaseException;

class Jabatan extends BaseController
{
    private JabatanModel $model;

    public function __construct()
    {
        $this->model = new JabatanModel();
    }

    private function withDepartemen(): JabatanModel
    {
        return $this->model
            ->select('jabatan.*, departemen.nama AS departemen_nama')
            ->join('departemen', 'departemen.id = jabatan.departemen_id', 'left');
    }

    private function detail($id): ?array
    {
        return $this->withDepartemen()->where('jabatan.id', $id)->first();
    }

    // GET /api/v1/jabatan?page=1&per_page=10&q=staf&departemen_id=1
    public function index()
    {
        $page    = max(1, (int) ($this->request->getGet('page') ?? 1));
        $perPage = min(100, max(1, (int) ($this->request->getGet('per_page') ?? 10)));
        $q       = trim((string) $this->request->getGet('q'));
        $deptId  = (int) ($this->request->getGet('departemen_id') ?? 0);

        $this->withDepartemen();

        if ($q !== '') {
            $this->model->like('jabatan.nama', $q);
        }
        if ($deptId > 0) {
            $this->model->where('jabatan.departemen_id', $deptId);
        }

        $total = $this->model->countAllResults(false);
        $rows  = $this->model->orderBy('jabatan.nama', 'ASC')->findAll($perPage, ($page - 1) * $perPage);

        return ApiResponse::success($rows, 'OK', [
            'page'        => $page,
            'per_page'    => $perPage,
            'total'       => $total,
            'total_pages' => (int) ceil($total / $perPage),
        ]);
    }

    public function show($id = null)
    {
        $row = $this->detail($id);
        if (! $row) {
            return ApiResponse::error('NOT_FOUND', 'Jabatan tidak ditemukan', null, 404);
        }
        return ApiResponse::success($row);
    }

    public function create()
    {
        $input = $this->request->getJSON(true) ?? [];

        $validation = service('validation');
        $validation->setRules($this->rules(null), $this->messages());

        if (! $validation->run($input)) {
            return ApiResponse::error('VALIDATION_ERROR', 'Data tidak valid', $validation->getErrors(), 422);
        }

        $id = $this->model->insert([
            'nama'          => trim($input['nama']),
            'departemen_id' => empty($input['departemen_id']) ? null : (int) $input['departemen_id'],
        ], true);

        return ApiResponse::success($this->detail($id), 'Jabatan dibuat', null, 201);
    }

    public function update($id = null)
    {
        if (! $this->model->find($id)) {
            return ApiResponse::error('NOT_FOUND', 'Jabatan tidak ditemukan', null, 404);
        }

        $input = $this->request->getJSON(true) ?? [];

        $validation = service('validation');
        $validation->setRules($this->rules((int) $id), $this->messages());

        if (! $validation->run($input)) {
            return ApiResponse::error('VALIDATION_ERROR', 'Data tidak valid', $validation->getErrors(), 422);
        }

        $this->model->update($id, [
            'nama'          => trim($input['nama']),
            'departemen_id' => empty($input['departemen_id']) ? null : (int) $input['departemen_id'],
        ]);

        return ApiResponse::success($this->detail($id), 'Jabatan diperbarui');
    }

    public function delete($id = null)
    {
        if (! $this->model->find($id)) {
            return ApiResponse::error('NOT_FOUND', 'Jabatan tidak ditemukan', null, 404);
        }

        try {
            $this->model->delete($id);
        } catch (DatabaseException $e) {
            return ApiResponse::error('IN_USE', 'Jabatan masih dipakai karyawan', null, 409);
        }

        return ApiResponse::success(null, 'Jabatan dihapus');
    }

    private function rules(?int $id): array
    {
        $unique = $id === null ? 'is_unique[jabatan.nama]' : 'is_unique[jabatan.nama,id,' . $id . ']';

        return [
            'nama'          => 'required|min_length[2]|max_length[100]|' . $unique,
            'departemen_id' => 'permit_empty|is_natural_no_zero|is_not_unique[departemen.id]',
        ];
    }

    private function messages(): array
    {
        return [
            'nama' => [
                'required'   => 'Nama jabatan wajib diisi',
                'min_length' => 'Nama minimal 2 karakter',
                'max_length' => 'Nama maksimal 100 karakter',
                'is_unique'  => 'Nama jabatan sudah dipakai',
            ],
            'departemen_id' => [
                'is_natural_no_zero' => 'Departemen tidak valid',
                'is_not_unique'      => 'Departemen tidak ditemukan',
            ],
        ];
    }
}

<?php

namespace App\Libraries;

use Firebase\JWT\JWT;
use Firebase\JWT\Key;

class JwtService
{
    /** User yang sedang login (diisi oleh AuthFilter). */
    public static ?array $user = null;

    public static function buat(array $user, int $ttl = 3600): string
    {
        $now = time();
        return JWT::encode([
            'iss'  => 'hadirpay',
            'iat'  => $now,
            'exp'  => $now + $ttl,
            'sub'  => (int) $user['id'],
            'role' => $user['role'],
        ], (string) env('JWT_SECRET'), 'HS256');
    }

    public static function baca(string $token): ?object
    {
        try {
            return JWT::decode($token, new Key((string) env('JWT_SECRET'), 'HS256'));
        } catch (\Throwable $e) {
            return null;
        }
    }
}

<?php

namespace App\Services;

use CodeIgniter\Database\BaseConnection;

class AttendanceService
{
    private BaseConnection $db;

    private const HARI = [1 => 'senin', 2 => 'selasa', 3 => 'rabu', 4 => 'kamis', 5 => 'jumat', 6 => 'sabtu', 7 => 'minggu'];

    public function __construct()
    {
        $this->db = db_connect();
    }

    // ---------- data pendukung ----------

    public function jadwal(): array
    {
        return $this->db->table('jadwal_kerja')->orderBy('is_default', 'DESC')->orderBy('id')->get(1)->getRowArray()
            ?? ['jam_masuk' => '08:00:00', 'jam_pulang' => '17:00:00', 'toleransi_menit' => 0, 'hari_kerja' => 'senin,selasa,rabu,kamis,jumat'];
    }

    public function lokasi(): ?array
    {
        return $this->db->table('lokasi_absensi')->where('aktif', 1)->orderBy('id')->get(1)->getRowArray();
    }

    /** Minimal durasi kerja (menit) sebelum boleh check-out. Default 480 (8 jam). Diatur lewat tabel pengaturan. */
    public function minDurasiMenit(): int
    {
        $row = $this->db->table('pengaturan')->where('key', 'absensi.min_durasi_menit')->get()->getRowArray();
        if ($row) {
            $nilai = json_decode($row['value_json'], true);
            if (is_numeric($nilai) && $nilai >= 0) {
                return (int) $nilai;
            }
        }
        return 480;
    }

    private function absensi(int $karyawanId, string $tanggal): ?array
    {
        return $this->db->table('absensi')->where('karyawan_id', $karyawanId)->where('tanggal', $tanggal)->get()->getRowArray();
    }

    private function jam(?string $t): ?string
    {
        return $t === null ? null : substr($t, 0, 5);
    }

    private function haversine(float $lat1, float $lng1, float $lat2, float $lng2): float
    {
        $r    = 6371000;
        $dLat = deg2rad($lat2 - $lat1);
        $dLng = deg2rad($lng2 - $lng1);
        $a    = sin($dLat / 2) ** 2 + cos(deg2rad($lat1)) * cos(deg2rad($lat2)) * sin($dLng / 2) ** 2;
        return 2 * $r * asin(sqrt($a));
    }

    /** @return array{error?: array, dalam?: ?bool} */
    private function evaluasiLokasi(?array $lokasi, $lat, $lng): array
    {
        if (! $lokasi) {
            return ['error' => ['LOCATION_NOT_CONFIGURED', 'Lokasi kerja belum diatur. Hubungi admin HRD', 409, null]];
        }

        $wajib = (int) $lokasi['wajib_lokasi'] === 1;
        $punya = is_numeric($lat) && is_numeric($lng);

        if ($punya && (abs((float) $lat) > 90 || abs((float) $lng) > 180)) {
            return ['error' => ['VALIDATION_ERROR', 'Data tidak valid', 422, ['latitude' => 'Koordinat tidak valid']]];
        }
        if (! $punya) {
            return $wajib
                ? ['error' => ['VALIDATION_ERROR', 'Data tidak valid', 422, ['latitude' => 'Lokasi wajib dikirim. Aktifkan izin lokasi di browser']]]
                : ['dalam' => null];
        }

        $jarak = $this->haversine((float) $lat, (float) $lng, (float) $lokasi['latitude'], (float) $lokasi['longitude']);
        $dalam = $jarak <= (int) $lokasi['radius_meter'];

        if (! $dalam && $wajib) {
            return ['error' => ['OUTSIDE_RADIUS', 'Anda berada di luar radius lokasi kerja (' . round($jarak) . ' m dari kantor)', 422, null]];
        }
        return ['dalam' => $dalam];
    }

    // ---------- endpoint karyawan ----------

    public function hariIni(int $karyawanId): array
    {
        $now     = new \DateTimeImmutable('now');
        $tanggal = $now->format('Y-m-d');
        $jadwal  = $this->jadwal();
        $lokasi  = $this->lokasi();
        $abs     = $this->absensi($karyawanId, $tanggal);

        $min         = $this->minDurasiMenit();
        $pulangCepat = ($abs && $abs['jam_masuk'] !== null)
            ? (new \DateTimeImmutable($tanggal . ' ' . $abs['jam_masuk']))->modify('+' . $min . ' minutes')
            : null;

        return [
            'tanggal'     => $tanggal,
            'server_time' => $now->format('c'),
            'jam_kerja'   => [
                'masuk'           => $this->jam($jadwal['jam_masuk']),
                'pulang'          => $this->jam($jadwal['jam_pulang']),
                'toleransi_menit' => (int) $jadwal['toleransi_menit'],
            ],
            'lokasi_kantor' => $lokasi ? [
                'nama'         => $lokasi['nama'],
                'alamat'       => $lokasi['alamat'],
                'latitude'     => (float) $lokasi['latitude'],
                'longitude'    => (float) $lokasi['longitude'],
                'radius_meter' => (int) $lokasi['radius_meter'],
                'wajib_lokasi' => (bool) $lokasi['wajib_lokasi'],
            ] : null,
            'absensi' => $abs ? [
                'id'              => (int) $abs['id'],
                'jam_masuk'       => $abs['jam_masuk'],
                'jam_pulang'      => $abs['jam_pulang'],
                'status'          => $abs['status'],
                'menit_terlambat' => (int) $abs['menit_terlambat'],
                'lokasi_masuk'    => [
                    'latitude'     => $abs['lokasi_masuk_lat'] !== null ? (float) $abs['lokasi_masuk_lat'] : null,
                    'longitude'    => $abs['lokasi_masuk_lng'] !== null ? (float) $abs['lokasi_masuk_lng'] : null,
                    'dalam_radius' => $abs['dalam_radius'] !== null ? (bool) $abs['dalam_radius'] : null,
                ],
            ] : null,
            'bisa_absen_masuk'    => $abs === null,
            'bisa_absen_pulang'   => $abs !== null && $abs['jam_masuk'] !== null && $abs['jam_pulang'] === null && $now >= $pulangCepat,
            'min_durasi_menit'    => $min,
            'pulang_paling_cepat' => $pulangCepat?->format('c'),
        ];
    }

    public function checkIn(int $karyawanId, $lat, $lng): array
    {
        $now     = new \DateTimeImmutable('now');
        $tanggal = $now->format('Y-m-d');
        $jam     = $now->format('H:i:s');

        $abs = $this->absensi($karyawanId, $tanggal);
        if ($abs) {
            return $abs['jam_masuk'] !== null
                ? ['error' => ['ALREADY_CHECKED_IN', 'Anda sudah absen masuk hari ini', 409, null]]
                : ['error' => ['CONFLICT', 'Hari ini Anda tercatat ' . $abs['status'], 409, null]];
        }

        $lokasi = $this->lokasi();
        $ev     = $this->evaluasiLokasi($lokasi, $lat, $lng);
        if (isset($ev['error'])) {
            return $ev;
        }

        $jadwal    = $this->jadwal();
        $selisih   = $now->getTimestamp() - strtotime($tanggal . ' ' . $jadwal['jam_masuk']);
        $terlambat = $selisih > ((int) $jadwal['toleransi_menit']) * 60;
        $menit     = $terlambat ? intdiv($selisih, 60) : 0;
        $status    = $terlambat ? 'terlambat' : 'hadir';
        $ts        = $now->format('Y-m-d H:i:s');

        $this->db->table('absensi')->insert([
            'karyawan_id'       => $karyawanId,
            'tanggal'           => $tanggal,
            'jam_masuk'         => $jam,
            'status'            => $status,
            'menit_terlambat'   => $menit,
            'lokasi_masuk_lat'  => is_numeric($lat) ? $lat : null,
            'lokasi_masuk_lng'  => is_numeric($lng) ? $lng : null,
            'dalam_radius'      => $ev['dalam'] === null ? null : (int) $ev['dalam'],
            'lokasi_absensi_id' => $lokasi['id'] ?? null,
            'sumber'            => 'web',
            'created_at'        => $ts,
            'updated_at'        => $ts,
        ]);

        return ['data' => [
            'id'              => (int) $this->db->insertID(),
            'tanggal'         => $tanggal,
            'jam_masuk'       => $jam,
            'status'          => $status,
            'menit_terlambat' => $menit,
        ]];
    }

    public function checkOut(int $karyawanId, $lat, $lng): array
    {
        $now     = new \DateTimeImmutable('now');
        $tanggal = $now->format('Y-m-d');

        $abs = $this->absensi($karyawanId, $tanggal);
        if (! $abs || $abs['jam_masuk'] === null) {
            return ['error' => ['NOT_CHECKED_IN', 'Anda belum absen masuk hari ini', 409, null]];
        }
        if ($abs['jam_pulang'] !== null) {
            return ['error' => ['ALREADY_CHECKED_OUT', 'Anda sudah absen pulang hari ini', 409, null]];
        }

        // Minimal durasi kerja sebelum boleh pulang (dapat diatur lewat tabel pengaturan)
        $min   = $this->minDurasiMenit();
        $sudah = intdiv($now->getTimestamp() - strtotime($tanggal . ' ' . $abs['jam_masuk']), 60);
        if ($sudah < $min) {
            $sisa = $min - $sudah;
            return ['error' => ['MIN_DURATION_NOT_MET', sprintf('Belum bisa absen pulang. Sisa waktu kerja minimal: %d jam %d menit', intdiv($sisa, 60), $sisa % 60), 422, null]];
        }

        $ev = $this->evaluasiLokasi($this->lokasi(), $lat, $lng);
        if (isset($ev['error'])) {
            return $ev;
        }

        $jadwal = $this->jadwal();
        $durasi = max(0, intdiv($now->getTimestamp() - strtotime($tanggal . ' ' . $abs['jam_masuk']), 60));
        $lembur = max(0, intdiv($now->getTimestamp() - strtotime($tanggal . ' ' . $jadwal['jam_pulang']), 60));
        $jam    = $now->format('H:i:s');

        $this->db->table('absensi')->where('id', $abs['id'])->update([
            'jam_pulang'        => $jam,
            'durasi_menit'      => $durasi,
            'lembur_menit'      => $lembur,
            'lokasi_pulang_lat' => is_numeric($lat) ? $lat : null,
            'lokasi_pulang_lng' => is_numeric($lng) ? $lng : null,
            'updated_at'        => $now->format('Y-m-d H:i:s'),
        ]);

        return ['data' => [
            'id'               => (int) $abs['id'],
            'jam_masuk'        => $abs['jam_masuk'],
            'jam_pulang'       => $jam,
            'durasi_menit'     => $durasi,
            'jam_lembur_menit' => $lembur,
        ]];
    }

    public function riwayat(int $karyawanId, string $periode, int $page, int $perPage): array
    {
        $awal  = $periode . '-01';
        $akhir = date('Y-m-t', strtotime($awal));

        $rows = $this->db->table('absensi')
            ->where('karyawan_id', $karyawanId)
            ->where('tanggal >=', $awal)->where('tanggal <=', $akhir)
            ->orderBy('tanggal', 'DESC')->get()->getResultArray();

        $ringkasan = ['hari_kerja' => $this->hitungHariKerja($awal, $akhir),
            'hadir' => 0, 'terlambat' => 0, 'izin' => 0, 'sakit' => 0, 'cuti' => 0, 'alpha' => 0];
        foreach ($rows as $r) {
            if (in_array($r['status'], ['hadir', 'terlambat', 'izin', 'sakit', 'cuti', 'alpha'], true)) {
                $ringkasan[$r['status']]++;
            }
        }

        $kalender = array_map(static fn ($r) => ['tanggal' => $r['tanggal'], 'status' => $r['status']], array_reverse($rows));

        $total   = count($rows);
        $slice   = array_slice($rows, ($page - 1) * $perPage, $perPage);
        $riwayat = array_map(fn ($r) => [
            'id'                   => (int) $r['id'],
            'tanggal'              => $r['tanggal'],
            'jam_masuk'            => $this->jam($r['jam_masuk']),
            'jam_pulang'           => $this->jam($r['jam_pulang']),
            'durasi_menit'         => $r['durasi_menit'] !== null ? (int) $r['durasi_menit'] : null,
            'status'               => $r['status'],
            'koreksi_pengajuan_id' => $r['pengajuan_id'] !== null ? (int) $r['pengajuan_id'] : null,
        ], $slice);

        return [
            'data' => ['periode' => $periode, 'ringkasan' => $ringkasan, 'kalender' => $kalender, 'riwayat' => $riwayat],
            'meta' => ['page' => $page, 'per_page' => $perPage, 'total' => $total, 'total_pages' => max(1, (int) ceil($total / $perPage))],
        ];
    }

    /** Hitung hari kerja menurut jadwal default, dikurangi hari libur. */
    public function hitungHariKerja(string $awal, string $akhir): int
    {
        $hariKerja = array_map('trim', explode(',', strtolower($this->jadwal()['hari_kerja'])));
        $libur     = array_column(
            $this->db->table('hari_libur')->select('tanggal')->where('tanggal >=', $awal)->where('tanggal <=', $akhir)->get()->getResultArray(),
            'tanggal'
        );

        $n = 0;
        for ($d = new \DateTimeImmutable($awal); $d->format('Y-m-d') <= $akhir; $d = $d->modify('+1 day')) {
            if (in_array(self::HARI[(int) $d->format('N')], $hariKerja, true) && ! in_array($d->format('Y-m-d'), $libur, true)) {
                $n++;
            }
        }
        return $n;
    }
